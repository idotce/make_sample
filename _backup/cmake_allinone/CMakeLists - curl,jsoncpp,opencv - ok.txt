cmake_minimum_required(VERSION 3.10)
project(Test)

# 设置C++标准
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

if(MSVC)
    set(CMAKE_CXX_FLAGS_RELEASE "${CMAKE_CXX_FLAGS_RELEASE} /MT")
    set(CMAKE_CXX_FLAGS_DEBUG "${CMAKE_CXX_FLAGS_DEBUG} /MTd")
endif()

# 强制编译静态库（全局设置）
set(BUILD_STATIC_LIBS ON CACHE BOOL "Build static libraries" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build shared libraries" FORCE)

# 设置第三方库目录
set(THIRD_PARTY_DIR ${CMAKE_SOURCE_DIR}/third_party)

# 保存当前状态
set(OLD_CMAKE_POLICY_DEFAULT_CMP0077 "$ENV{CMAKE_POLICY_DEFAULT_CMP0077}")

# ======================== curl 配置 ========================
set(CURL_SOURCE_DIR ${THIRD_PARTY_DIR}/curl)
set(CURL_BINARY_DIR ${CMAKE_BINARY_DIR}/third_party/curl)
file(MAKE_DIRECTORY ${CURL_BINARY_DIR})

set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(CURL_STATICLIB ON CACHE BOOL "" FORCE)

add_subdirectory(${CURL_SOURCE_DIR} ${CURL_BINARY_DIR} EXCLUDE_FROM_ALL)
include_directories(${CURL_SOURCE_DIR}/include)

# ======================== jsoncpp 配置 ========================
set(JSONCPP_SOURCE_DIR ${THIRD_PARTY_DIR}/jsoncpp)
set(JSONCPP_BINARY_DIR ${CMAKE_BINARY_DIR}/third_party/jsoncpp)
file(MAKE_DIRECTORY ${JSONCPP_BINARY_DIR})

set(JSONCPP_WITH_TESTS OFF CACHE BOOL "" FORCE)
set(JSONCPP_STATIC_WINDOWS_RUNTIME ON CACHE BOOL "" FORCE)

add_subdirectory(${JSONCPP_SOURCE_DIR} ${JSONCPP_BINARY_DIR} EXCLUDE_FROM_ALL)
include_directories(${JSONCPP_SOURCE_DIR}/include)

# ======================== OpenCV 配置 ========================
# 在编译 OpenCV 之前，恢复策略设置
if(OLD_CMAKE_POLICY_DEFAULT_CMP0077)
    set(ENV{CMAKE_POLICY_DEFAULT_CMP0077} ${OLD_CMAKE_POLICY_DEFAULT_CMP0077})
endif()

set(OpenCV_SOURCE_DIR ${THIRD_PARTY_DIR}/opencv)
set(OpenCV_BINARY_DIR ${CMAKE_BINARY_DIR}/third_party/opencv)
set(OpenCV_LIB_DIR ${OpenCV_BINARY_DIR}/lib/Release)  # 显式指定OpenCV库输出目录
file(MAKE_DIRECTORY ${OpenCV_BINARY_DIR})

set(BUILD_opencv_world ON CACHE BOOL "" FORCE)
set(OPENCV_FORCE_3RDPARTY_BUILD ON CACHE BOOL "" FORCE)
# set(WITH_IPP ON CACHE BOOL "" FORCE)
# set(OPENCV_IPP ON CACHE BOOL "" FORCE)
# set(ENABLE_IPPICV ON CACHE BOOL "" FORCE)
# set(WITH_FFMPEG OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_PERF_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(BUILD_opencv_java OFF CACHE BOOL "" FORCE)
set(BUILD_opencv_python OFF CACHE BOOL "" FORCE)

add_subdirectory(${OpenCV_SOURCE_DIR} ${OpenCV_BINARY_DIR} EXCLUDE_FROM_ALL)
set(OpenCV_INCLUDE_DIRS
    ${OpenCV_SOURCE_DIR}/include
    ${OpenCV_BINARY_DIR}/include
    ${CMAKE_BINARY_DIR}
    ${THIRD_PARTY_DIR}/opencv/modules/calib3d/include
    ${THIRD_PARTY_DIR}/opencv/modules/core/include
    ${THIRD_PARTY_DIR}/opencv/modules/dnn/include
    ${THIRD_PARTY_DIR}/opencv/modules/features2d/include
    ${THIRD_PARTY_DIR}/opencv/modules/flann/include
    ${THIRD_PARTY_DIR}/opencv/modules/highgui/include
    ${THIRD_PARTY_DIR}/opencv/modules/imgcodecs/include
    ${THIRD_PARTY_DIR}/opencv/modules/imgproc/include
    ${THIRD_PARTY_DIR}/opencv/modules/ml/include
    ${THIRD_PARTY_DIR}/opencv/modules/objdetect/include
    ${THIRD_PARTY_DIR}/opencv/modules/photo/include
    ${THIRD_PARTY_DIR}/opencv/modules/stitching/include
    ${THIRD_PARTY_DIR}/opencv/modules/video/include
    ${THIRD_PARTY_DIR}/opencv/modules/videoio/include
)
include_directories(${OpenCV_INCLUDE_DIRS})

set(OpenCV_OUTPUT_LIB ${OpenCV_BINARY_DIR}/lib)
set(OpenCV_OUTPUT_BIN ${OpenCV_BINARY_DIR}/bin)
set_target_properties(opencv_world PROPERTIES
    ARCHIVE_OUTPUT_DIRECTORY ${OpenCV_OUTPUT_LIB}  # 静态库输出到 opencv/lib
    LIBRARY_OUTPUT_DIRECTORY ${OpenCV_OUTPUT_LIB}  # 动态库（若启用）输出到 opencv/lib
    RUNTIME_OUTPUT_DIRECTORY ${OpenCV_OUTPUT_BIN}  # 若 OpenCV 生成工具可执行文件，输出到 opencv/bin
    # 针对 MSVC 的 Debug/Release 多配置，确保不同配置产物分离（可选）
    ARCHIVE_OUTPUT_DIRECTORY_DEBUG ${OpenCV_OUTPUT_LIB}/Debug
    ARCHIVE_OUTPUT_DIRECTORY_RELEASE ${OpenCV_OUTPUT_LIB}/Release
    RUNTIME_OUTPUT_DIRECTORY_DEBUG ${OpenCV_OUTPUT_BIN}/Debug
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${OpenCV_OUTPUT_BIN}/Release
)

# ======================== 主程序配置 ========================
aux_source_directory(src SRC_LIST)
add_executable(${PROJECT_NAME} ${SRC_LIST})

# 链接库 - 使用目标名称而非库文件名，确保依赖关系正确
link_directories(${OpenCV_LIB_DIR})
target_link_libraries(${PROJECT_NAME}
    libcurl
    jsoncpp_static
    opencv_world
)

# 针对Windows的特殊配置
if(WIN32)
    target_link_libraries(${PROJECT_NAME} ws2_32 wldap32 crypt32)
    target_compile_definitions(${PROJECT_NAME} PRIVATE
        _CRT_SECURE_NO_WARNINGS
        CURL_STATICLIB
        OPENCV_STATIC  # 添加OpenCV静态链接宏
    )
    target_link_libraries(${PROJECT_NAME} ws2_32 wldap32 crypt32)
endif()

# 针对Linux的特殊配置
if(UNIX AND NOT APPLE)
    target_link_libraries(${PROJECT_NAME} pthread dl)
endif()

# utf8编码支持
if(MSVC)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${PROJECT_NAME} PRIVATE "/utf-8")
    endif()
endif()

# 安装配置
install(TARGETS ${PROJECT_NAME}
    RUNTIME DESTINATION bin
    CONFIGURATIONS Release
)
